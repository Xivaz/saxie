DECLARE
  CURSOR C_ENDR IS
   SELECT ENDR_IN_ID, 
          ENDR_IN_CEP, ENDR_ST_LOGRADOURO, ENDR_IN_NUMERO, 
          ENDR_ST_COMPLEMENTO, ENDR_ST_BAIRRO, ENDR_ST_UF, 
          MUNI_IN_CODIGO_CENSO
     FROM TBL_ENDERECO
    WHERE ENDR_ST_TIPO_REFERENCIA = :P49_ENDR_ST_TIPO_REFERENCIA
      AND ENDR_IN_ID_REFERENCIA = :P49_ENDR_IN_ID_REFERENCIA;
BEGIN
  OPEN C_ENDR;
  FETCH C_ENDR INTO :P49_ENDR_IN_ID, :P49_ENDR_IN_CEP,
                    :P49_ENDR_ST_LOGRADOURO, :P49_ENDR_IN_NUMERO,
                    :P49_ENDR_ST_COMPLEMENTO, :P49_ENDR_ST_BAIRRO,
                    :P49_ENDR_ST_UF, :P49_MUNI_IN_CODIGO_CENSO;
  CLOSE C_ENDR;
END;
